cmake_minimum_required(VERSION 3.20)
project(MaxPrioGames VERSION 1.0.0 LANGUAGES CXX)

set(MPG_VERSION "${PROJECT_VERSION}" CACHE STRING "Release version (major.minor.patch)")
if(NOT MPG_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
  message(FATAL_ERROR "MPG_VERSION must be major.minor.patch")
endif()
string(REPLACE "." "," MPG_VERSION_COMMA "${MPG_VERSION}")
configure_file(src/version.hpp.in generated/version.hpp @ONLY)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_library(mpg_policy STATIC src/policy.cpp src/catalog.cpp src/steam_cache.cpp)
target_include_directories(mpg_policy PUBLIC src)

include(CTest)
if(BUILD_TESTING)
  add_executable(catalog_test tests/catalog_test.cpp)
  target_link_libraries(catalog_test PRIVATE mpg_policy)
  add_test(NAME curated_catalog COMMAND catalog_test)
  add_executable(steam_cache_test tests/steam_cache_test.cpp)
  target_link_libraries(steam_cache_test PRIVATE mpg_policy)
  add_test(NAME steam_appinfo_classification COMMAND steam_cache_test)
  add_executable(backend_test tests/backend_test.cpp src/discovery.cpp)
  target_link_libraries(backend_test PRIVATE mpg_policy)
  target_include_directories(backend_test PRIVATE src)
  add_test(NAME process_matching_policy COMMAND backend_test)
  foreach(test_target catalog_test steam_cache_test backend_test)
    if(MSVC)
      target_compile_options(${test_target} PRIVATE /W4 /utf-8 /UNDEBUG)
    else()
      target_compile_options(${test_target} PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
    endif()
  endforeach()
endif()

if(WIN32)
  enable_language(RC)
  set(MPG_WINDOWS_DEFINITIONS UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX
    _WIN32_WINNT=0x0A00 WINVER=0x0A00 NTDDI_VERSION=0x0A000006)
  if(BUILD_TESTING)
    target_compile_definitions(backend_test PRIVATE ${MPG_WINDOWS_DEFINITIONS})
    target_link_libraries(backend_test PRIVATE advapi32 shell32 ole32 uuid)
    add_executable(config_test tests/config_test.cpp src/config.cpp)
    target_include_directories(config_test PRIVATE src)
    target_compile_definitions(config_test PRIVATE ${MPG_WINDOWS_DEFINITIONS})
    target_link_libraries(config_test PRIVATE mpg_policy shell32 ole32 advapi32 uuid)
    add_executable(backend_helper tests/backend_helper.cpp)
    target_compile_definitions(backend_helper PRIVATE ${MPG_WINDOWS_DEFINITIONS})
    add_executable(backend_windows_test tests/backend_windows_test.cpp src/engine.cpp src/discovery.cpp)
    target_include_directories(backend_windows_test PRIVATE src)
    target_compile_definitions(backend_windows_test PRIVATE ${MPG_WINDOWS_DEFINITIONS})
    target_link_libraries(backend_windows_test PRIVATE mpg_policy advapi32 shell32 ole32 uuid)
    add_dependencies(backend_windows_test backend_helper)
    foreach(test_target config_test backend_windows_test backend_helper)
      if(MSVC)
        target_compile_options(${test_target} PRIVATE /W4 /utf-8 /UNDEBUG)
      endif()
    endforeach()
    add_test(NAME settings_persistence COMMAND config_test)
    add_test(NAME process_priority_affinity COMMAND backend_windows_test "$<TARGET_FILE:backend_helper>")
    set_tests_properties(process_priority_affinity PROPERTIES TIMEOUT 60)
  endif()
  configure_file(src/app.rc.in generated/app.rc @ONLY)
  add_executable(MaxPrioGames WIN32 src/app.cpp src/config.cpp src/startup.cpp src/engine.cpp src/discovery.cpp
    "${CMAKE_CURRENT_BINARY_DIR}/generated/app.rc")
  target_include_directories(MaxPrioGames PRIVATE src "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_compile_definitions(MaxPrioGames PRIVATE ${MPG_WINDOWS_DEFINITIONS})
  target_link_libraries(MaxPrioGames PRIVATE mpg_policy
    user32 gdi32 comctl32 comdlg32 shell32 shlwapi advapi32
    ole32 oleaut32 uuid psapi taskschd)
  if(MSVC)
    target_compile_options(MaxPrioGames PRIVATE /W4 /utf-8 /permissive- /Zc:__cplusplus /EHsc /guard:cf)
    target_compile_options(mpg_policy PRIVATE /W4 /utf-8 /permissive-)
    foreach(target MaxPrioGames mpg_policy)
      target_compile_options(${target} PRIVATE $<$<CONFIG:Release>:/O1> $<$<CONFIG:Release>:/GL> $<$<CONFIG:Release>:/Gy> $<$<CONFIG:Release>:/Gw>)
    endforeach()
    target_link_options(MaxPrioGames PRIVATE /MANIFEST:NO /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf
      $<$<CONFIG:Release>:/LTCG> $<$<CONFIG:Release>:/OPT:REF> $<$<CONFIG:Release>:/OPT:ICF>)
  else()
    target_compile_options(MaxPrioGames PRIVATE -Wall -Wextra -Os -ffunction-sections -fdata-sections)
    target_link_options(MaxPrioGames PRIVATE -municode -static -static-libgcc -static-libstdc++ -Wl,--gc-sections)
  endif()
  install(TARGETS MaxPrioGames RUNTIME DESTINATION .)
else()
  message(STATUS "Non-Windows host: building portable matching/catalog/discovery tests only.")
endif()
